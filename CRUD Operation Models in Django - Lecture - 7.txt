from django.db import models

# Create your models here.

class Customer(models.Model): # Model Class
    name=models.CharField(max_length=50)
    age=models.IntegerField()
    mobileno=models.CharField(max_length=15)
    post=models.CharField(max_length=50)
    salary=models.DecimalField(max_digits=25,decimal_places=2)
    def __str__(self):
        return self.name

def view_customer_information(request):
    if request.method=='GET':
        return render(request,'CMS/customer.html')
    elif request.method=='POST':
        if 'btnInsert' in request.POST:
            cus=Customer()
            cus.name=request.POST.get('txtName','NA')
            cus.age=request.POST.get('txtAge','NA')
            cus.mobileno=request.POST.get('txtMob','NA')
            cus.post=request.POST.get('txtPost','NA')
            cus.salary=request.POST.get('txtSalary','NA')
            cus.save() 
            return HttpResponse("<h1>Record Inserted!!</h1>")
        elif 'btnSingle' in request.POST:
            cid=int(request.POST.get('txtCid',0))
            cus=Customer.objects.get(id=cid)
            d1={'cus':cus}
            return render(request,'CMS/customer.html',context=d1)
        elif 'btnUpdate' in request.POST:
            cus=Customer()
            cus.id=int(request.POST.get('txtCid',0))
            if Customer.objects.filter(id=cus.id).exists():
                cus.name=request.POST.get('txtName','NA')
                cus.age=request.POST.get('txtAge','NA')
                cus.mobileno=request.POST.get('txtMob','NA')
                cus.post=request.POST.get('txtPost','NA')
                cus.salary=request.POST.get('txtSalary','NA')
                cus.save() 
                return HttpResponse("<h1>Record Updated!!</h1>")
        elif 'btnDelete' in request.POST:
            cus=Customer()
            cus.id=int(request.POST.get('txtCid',0))
            Customer.objects.filter(id=cus.id).delete()
            return HttpResponse("<h1>Record Deleted!!</h1>")
        elif 'btnShowAll' in request.POST:
            cusall=Customer.objects.all()
            d1={'customers':cusall}
            return render(request,'CMS/customer.html',context=d1)
