class Customer:
    def __init__(self):
        self.cid=0
        self.name=''
        self.age=0
        self.mobileno=''
        self.city=''
def getCustomer(N):
    cusList=[]
    for i in range(1,N+1):
        cus=Customer()
        cus.cid=100 +i
        cus.name='Nikhil'+str(i)
        cus.age=21+i
        cus.mobileno='979879911'+str(i)
        cus.city='Noida'+str(i)
        cusList.append(cus)
    return cusList

def view_dtl(request):
    cno=int(request.POST.get('txtNo',0))
    customers=getCustomer(cno)
    d1={'customers':customers}
    return render(request,'CMS/dtl.html',context=d1)

from django.db import models

# Create your models here.

class Customer(models.Model): # Model Class
    #cid=models.IntegerField() It will create automatically.
    name=models.CharField(max_length=50)
    age=models.IntegerField()
    mobileno=models.CharField(max_length=15)
    post=models.CharField(max_length=50)
    salary=models.DecimalField(max_digits=25,decimal_places=2)
    def __str__(self):
        return self.name
