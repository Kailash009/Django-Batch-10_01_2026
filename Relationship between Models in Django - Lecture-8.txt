Relationship between Models in Django:-
------------------------------------------------
There are three types of relationship between models:-
a) One to One.
b) One to Many or Many to one.
c) Many to Many.
-----------------------------------------------------------------
Student -  Personal Detail  =>>> One to One.
Student - PaymentDetails => One to Many or Many to one.
Student - Course  => Many to Many.
1 S => Many Course.
1 C => Many Student.

from django.shortcuts import render
from SMS.models import *

# Create your views here.

def view_get_student_payments(request):
    if request.method=='GET':
        return render(request,'SMS/payment.html')
    elif request.method=='POST':
        sid=int(request.POST.get('txtSid',0))
        stu=Student.objects.get(id=sid)
        allpayment=stu.paymentdetails_set.all()
        d1={'payments':allpayment,'stu':stu}
        return render(request,'SMS/payment.html',context=d1)

def view_get__coursewise_students(request):
    courses=Course.objects.all()
    d1={'courses':courses}
    if request.method=='GET':  
        cid=Course.objects.get(id=1)
        d1['cid']=cid
        d1['cname']=cid.name
        allstu=cid.students.all()
        d1['students']=allstu
        return render(request,'SMS/student.html',context=d1)
    elif request.method=='POST':
        cid=int(request.POST.get('Course',0))
        course=Course.objects.get(id=cid)
        d1['cid']=course.id
        d1['cname']=course.name
        allstu=course.students.all()
        d1['students']=allstu
        return render(request,'SMS/student.html',context=d1)
